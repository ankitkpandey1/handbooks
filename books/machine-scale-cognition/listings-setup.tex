% The listings package typesets code in the surrounding body font unless told
% otherwise. This book's body font (Noto Serif) has no box-drawing, arrow, or
% minus-sign glyphs, and its proportional widths destroy column alignment in
% ASCII diagrams — so every code block and inline code span must use the
% declared monofont instead.
\lstset{%
  basicstyle=\small\ttfamily,
  columns=fullflexible,
  keepspaces=true,
  breaklines=true,
  breakatwhitespace=true,
  showstringspaces=false,
}

% The article class gives subsection numbers a 2.3em column in the contents
% page, which two-digit numbers such as 11.10 overrun into their titles.
\makeatletter
\renewcommand*\l@subsection{\@dottedtocline{2}{1.5em}{3.2em}}
\makeatother
